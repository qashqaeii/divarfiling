# حجم دیتابیس
sudo -u postgres psql -d divarfilingdb -c "

"

# تعداد Connectionها
sudo -u postgres psql -d divarfilingdb -c "
SELECT state, COUNT(*)
FROM pg_stat_activity
WHERE datname='divarfilingdb'
GROUP BY state;
"
